Architecture SQL
Base de données PostgreSQL complète — toutes les données utilisateurs, transactions, codes et logs sont stockées, trackées et auditables pour les investisseurs.
TABLE: users
CREATE TABLE users (
  id             UUID          PRIMARY KEY DEFAULT gen_random_uuid(),
  first_name     VARCHAR(100)  NOT NULL,
  last_name      VARCHAR(100)  NOT NULL,
  email          VARCHAR(255)  UNIQUE NOT NULL,
  phone          VARCHAR(25)   UNIQUE NOT NULL,
  country_code   CHAR(2)       NOT NULL,
  date_of_birth  DATE          NOT NULL,
  password_hash  TEXT          NOT NULL,   -- bcrypt
  pin_hash       TEXT          NOT NULL,   -- bcrypt
  kyc_status     VARCHAR(20)   DEFAULT 'pending',  -- pending|verified|rejected
  kyc_doc_type   VARCHAR(50),
  kyc_doc_num    VARCHAR(50),
  kyc_verified_at TIMESTAMP,
  balance_usd    DECIMAL(15,4) DEFAULT 0.0000,
  card_number    TEXT,                        -- AES-256 encrypted
  card_expiry    DATE,
  card_cvv_hash  TEXT,                        -- hashed
  card_frozen    BOOLEAN       DEFAULT FALSE,
  daily_limit    DECIMAL(10,2) DEFAULT 500.00,
  is_active      BOOLEAN       DEFAULT TRUE,
  last_login     TIMESTAMP,
  created_at     TIMESTAMP     DEFAULT NOW(),
  updated_at     TIMESTAMP     DEFAULT NOW()
);

CREATE INDEX idx_users_email   ON users(email);
CREATE INDEX idx_users_phone   ON users(phone);
CREATE INDEX idx_users_country ON users(country_code);
TABLE: transactions
CREATE TABLE transactions (
  id              UUID          PRIMARY KEY DEFAULT gen_random_uuid(),
  ref             VARCHAR(30)   UNIQUE NOT NULL,  -- FP-2024-XXXXX
  user_id         UUID          REFERENCES users(id) ON DELETE SET NULL,
  user_name       VARCHAR(200),
  type            VARCHAR(30)   NOT NULL,  -- recharge|conversion|card_payment|refund
  operator        VARCHAR(20),              -- mtn|orange|airtel
  amount_local    DECIMAL(15,2),
  currency_local  CHAR(3),
  amount_usd      DECIMAL(15,4),
  target_currency CHAR(3)       DEFAULT 'USD',
  exchange_rate   DECIMAL(12,6),
  fee             DECIMAL(10,4) DEFAULT 0,
  status          VARCHAR(20)   DEFAULT 'pending', -- pending|completed|failed|flagged
  country_code    CHAR(2),
  phone_number    VARCHAR(25),
  ip_address      INET,
  device_hash     TEXT,
  completed_at    TIMESTAMP,
  created_at      TIMESTAMP     DEFAULT NOW()
);

CREATE INDEX idx_txn_user    ON transactions(user_id);
CREATE INDEX idx_txn_status  ON transactions(status);
CREATE INDEX idx_txn_country ON transactions(country_code);
CREATE INDEX idx_txn_created ON transactions(created_at DESC);
TABLE: payment_codes
CREATE TABLE payment_codes (
  id              UUID         PRIMARY KEY DEFAULT gen_random_uuid(),
  transaction_id  UUID         REFERENCES transactions(id) ON DELETE CASCADE,
  code            VARCHAR(8)   NOT NULL,    -- 6 chiffres, usage unique
  expires_at      TIMESTAMP    NOT NULL,    -- created_at + 24h
  is_used         BOOLEAN      DEFAULT FALSE,
  used_at         TIMESTAMP,
  created_at      TIMESTAMP    DEFAULT NOW(),
  CONSTRAINT chk_expiry CHECK (expires_at > created_at)
);

-- Index pour purge automatique (pg_cron toutes les heures)
CREATE INDEX idx_code_expiry ON payment_codes(expires_at);

TABLE: fraud_logs
CREATE TABLE fraud_logs (
  id              UUID         PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         UUID         REFERENCES users(id),
  transaction_id  UUID         REFERENCES transactions(id),
  risk_score      SMALLINT     CHECK (risk_score BETWEEN 0 AND 100),
  flag_reason     TEXT,
  flag_type       VARCHAR(50),  -- unusual_amount|new_country|high_freq|velocity
  reviewed_by     UUID         REFERENCES admins(id),
  reviewed_at     TIMESTAMP,
  action_taken    VARCHAR(20)  DEFAULT 'none', -- none|approved|blocked|escalated
  notes           TEXT,
  created_at      TIMESTAMP    DEFAULT NOW()
);
TABLE: admins
CREATE TABLE admins (
  id             UUID         PRIMARY KEY DEFAULT gen_random_uuid(),
  email          VARCHAR(255) UNIQUE NOT NULL,
  password_hash  TEXT         NOT NULL,
  role           VARCHAR(30)  DEFAULT 'support', -- super_admin|compliance|support
  permissions    JSONB        DEFAULT '{}'::jsonb,
  is_active      BOOLEAN      DEFAULT TRUE,
  last_login     TIMESTAMP,
  created_at     TIMESTAMP    DEFAULT NOW()
);
TABLE: exchange_rates
CREATE TABLE exchange_rates (
  id            UUID         PRIMARY KEY DEFAULT gen_random_uuid(),
  from_currency CHAR(3)      NOT NULL,   -- XAF, GNF, NGN...
  to_currency   CHAR(3)      NOT NULL,   -- USD, EUR, GBP...
  rate          DECIMAL(12,6) NOT NULL,
  source        VARCHAR(50),              -- ECB, OpenExchange...
  recorded_at   TIMESTAMP    DEFAULT NOW()
);

CREATE INDEX idx_rates_pair ON exchange_rates(from_currency, to_currency);
CREATE INDEX idx_rates_time ON exchange_rates(recorded_at DESC);
